            FastAPI middleware
            
          
    
          
    from pyllm_guard import Guard
from fastapi import FastAPI, Request

guard = Guard()

app = FastAPI()

@app.middleware("http")
async def scan_prompts(request: Request, call_next):
    prompt = await request.body()
    guard.scan(prompt)
    return await call_next(request)
